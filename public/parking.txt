  /**
   * Creates the floor headings in the dom and prepares them for user
   * interaction
   */
  function createInitial() {
    const floors = ["E18"];
    for (let i = 0; i < floors.length; i++) {
      let floorHeading = document.createElement("h2");
      floorHeading.textContent = floors[i];
      floorHeading.id = floors[i];
      floorHeading.classList.add("floor");
      document.getElementById("levels").appendChild(floorHeading);
      floorHeading.addEventListener("click", switchLevel);
    }
  }

  function switchLevel() {

    if (oldFloor !== null) {
      oldFloor.classList.replace("selected-floor", "floor");
    }
    const garage = createGarageInfo();

  /**
   * This function takes an id of a floor and converts it into an index.
   * Returns -1 if the given id is not associated with a floor
   * @param {String} id given id of floor
   * @param {Array} garage array representing garage
   * @returns {Integer} a number to represent the index of the given floor id
   */
  function findFloorIndex(id, garage) {
    for (let i = 0; i < garage.length; i++) {
      if (garage[i][0] === id) {
        return i;
      }
    }
    return -1;
  }
